class PythonSource(BaseModel):
    """Validated Python source code — rejects syntactically invalid code."""

    code: str = Field(description="Python source code")

    @field_validator("code")
    @classmethod
    def validate_python_syntax(cls, v: str) -> str:
        if not v.strip():
            return v
        try:
            ast.parse(v)
        except SyntaxError as e:
            raise ValueError(f"Invalid Python syntax: {e}") from e
        return v

    def __str__(self) -> str:
        return self.code


class SolverInput(BaseModel):
    """Input for one solver run."""

    question: str = Field(description="The business question to answer")
    gt_answer: str = Field(default="", description="Ground-truth answer (bootstrap only)")
    guidelines: str = Field(default="", description="Answer format guidelines")
    existing_functions: str = Field(default="", description="Helper functions from skill library")
    existing_examples: str = Field(default="", description="Worked examples from skill library")


class DataContext(BaseModel):
    """Pre-loaded data files shared across all sub-agents."""

    model_config = {"arbitrary_types_allowed": True}

    text_files: dict[str, str] = Field(default_factory=dict)
    json_files: dict[str, Any] = Field(default_factory=dict)
    dataframes: dict[str, Any] = Field(default_factory=dict)

    def summary(self) -> str:
        lines = []
        for name, df in sorted(self.dataframes.items()):
            lines.append(f"- dataframes['{name}']: {df.shape[0]:,} rows x {df.shape[1]} cols")
        for name, data in sorted(self.json_files.items()):
            count = len(data) if isinstance(data, (list, dict)) else "?"
            lines.append(f"- json_files['{name}']: {count} items")
        for name in sorted(self.text_files.keys()):
            lines.append(f"- text_files['{name}']")
        return "\n".join(lines)

    @classmethod
    def from_dir(cls, data_dir: str) -> "DataContext":
        """Load all CSV, JSON, and text files from a directory."""
        ctx = cls()
        if not os.path.isdir(data_dir):
            return ctx
        for filename in os.listdir(data_dir):
            filepath = os.path.join(data_dir, filename)
            if os.path.isdir(filepath):
                continue
            _, ext = os.path.splitext(filename)
            if ext in {".md", ".txt", ".rst"}:
                with open(filepath) as f:
                    ctx.text_files[filename] = f.read()
            elif ext == ".json":
                with open(filepath) as f:
                    ctx.json_files[filename] = json.load(f)
            elif ext == ".csv":
                ctx.dataframes[filename] = pd.read_csv(filepath)
        return ctx

            "Python source code with reusable helper function definitions. "
            "Must encode domain logic (fee calculations, matching rules). "
            "Include type hints, docstrings, inline imports. "
            "Empty string if no generalizable functions found."
        )
    )
    example_name: str = Field(
        description="Short descriptive name, e.g. 'Finding Applicable Fee IDs for a Merchant'"
    )
    example_question_pattern: str = Field(
        description="Generalized pattern with <placeholders>, e.g. 'What fees apply to <merchant>?'"
    )
    example_rule: str = Field(
        description="Key business rule or domain insight needed to solve this type of problem"
    )
    example_code: str = Field(
        description=(
            "3-10 lines of executable Python showing how to solve this type. "
            "Real code (not pseudocode), copy-paste ready."
        )
    )
    solution_script: PythonSource | None = Field(
        default=None,
        description="Standalone reproducible Python script for this specific task (bootstrap only)",

# ---------------------------------------------------------------------------
# Fee Matching Helpers (available to LLM-generated code at runtime)
# ---------------------------------------------------------------------------


def fee_applies_to_all(value: Any) -> bool:
    """True if fee field means 'applies to all'. Both None AND [] match everything."""
    return value is None or value == []


def volume_matches(fee_volume: str | None, actual_volume: float) -> bool:
    """Check if actual monthly volume satisfies fee's monthly_volume constraint."""
    if fee_volume is None:
        return True
    if fee_volume == "<100k":
        return actual_volume < 100_000
    if fee_volume == "100k-1m":
        return 100_000 <= actual_volume < 1_000_000
    if fee_volume == "1m-5m":
        return 1_000_000 <= actual_volume < 5_000_000
    if fee_volume == ">5m":
        return actual_volume >= 5_000_000
    return False


def fraud_level_matches(fee_fraud: str | None, actual_fraud_pct: float) -> bool:
    """Check if actual fraud % satisfies fee's monthly_fraud_level constraint."""
    if fee_fraud is None:
        return True
    if fee_fraud == "<7.2%":
        return actual_fraud_pct < 7.2
    if fee_fraud == "7.2%-7.7%":
        return 7.2 <= actual_fraud_pct < 7.7
    if fee_fraud == "7.7%-8.3%":
        return 7.7 <= actual_fraud_pct < 8.3
    if fee_fraud == ">8.3%":
        return actual_fraud_pct >= 8.3
    return False


def calc_fee(fee: dict, transaction_amount: float) -> float:
    """fee = fixed_amount + rate * amount / 10000"""
    return fee["fixed_amount"] + fee["rate"] * transaction_amount / 10_000


def find_lowest_fee(matching_fees: list[dict], transaction_amount: float) -> dict | None:
    """Return the fee with the lowest calculated amount. None if list is empty."""
    if not matching_fees:
        return None
    return min(matching_fees, key=lambda f: calc_fee(f, transaction_amount))
```

# ---------------------------------------------------------------------------
# SkillLibrary — manages skills/zero_shot_examples.py
# ---------------------------------------------------------------------------

import importlib.util
import textwrap


class SkillLibrary:
    """Load and save accumulated skills to/from zero_shot_examples.py."""

    FILENAME = "zero_shot_examples.py"

    def __init__(self, skills_dir: Path | str):
        self.skills_dir = Path(skills_dir)

    @property
    def _skills_file(self) -> Path:
        return self.skills_dir / self.FILENAME

    def load(self) -> tuple[str, str]:
        """Return (helper_functions_str, worked_examples_str) from the skill file.
        Returns ("", "") if the file does not exist or has no content.
        """
        if not self._skills_file.exists():
            return "", ""
        try:
            spec = importlib.util.spec_from_file_location("_zse", str(self._skills_file))
            mod = importlib.util.module_from_spec(spec)
            spec.loader.exec_module(mod)
            funcs = getattr(mod, "HELPER_FUNCTIONS", "") or ""
            examples = getattr(mod, "get_examples_prompt", lambda: "")() or ""
            return funcs, examples
        except Exception:
            return "", ""

    def save(self, extraction: ExtractionResult) -> None:
        """Merge a new ExtractionResult into zero_shot_examples.py.

        - Appends helper_functions (deduplicated by function name).
        - Appends the worked example, replacing any with the same example_name.
        """
        # Load existing
        funcs_str, _ = self.load()

        existing_examples: list[str] = []
        if self._skills_file.exists():
            try:
                spec = importlib.util.spec_from_file_location("_zse2", str(self._skills_file))
                mod = importlib.util.module_from_spec(spec)
                spec.loader.exec_module(mod)
                existing_examples = list(getattr(mod, "EXAMPLES", []))
            except Exception:
                pass

        # Merge helper functions (simple string append, avoid exact duplicates)
        new_funcs = funcs_str
        if extraction.helper_functions and "def " in extraction.helper_functions:
            if extraction.helper_functions not in funcs_str:
                new_funcs = (funcs_str + "\n\n" + extraction.helper_functions).strip()

        # Build new example entry
        if extraction.example_code.strip():
            new_example = (
                f"EXAMPLE {len(existing_examples) + 1} - {extraction.example_name}:\n"
                f"Q: {extraction.example_question_pattern}\n"
                f"Rule: {extraction.example_rule}\n"
                f"A:\n{extraction.example_code.strip()}"
            )
        else:
            new_example = None

        # Deduplicate examples by name
        merged = [
            ex for ex in existing_examples
            if extraction.example_name not in ex
        ]
        if new_example:
            merged.append(new_example)

        # Write file
        lines = [
            "# Generated by DABstep bootstrap/discover pipeline.",
            "# Do not edit manually — re-run bootstrap to update.",
            "",
            f'HELPER_FUNCTIONS = {repr(new_funcs)}',
            "",
            "EXAMPLES: list[str] = [",
        ]
        for ex in merged:
            lines.append(f"    {repr(ex)},")
        lines += [
            "]",
            "",
            "",
            "def get_examples_prompt() -> str:",
            '    """Return all worked examples formatted for system prompt inclusion."""',
            '    return "\\n\\n".join(EXAMPLES) if EXAMPLES else "(No examples yet.)"',
        ]
        self._skills_file.write_text("\n".join(lines) + "\n")
```

# ---------------------------------------------------------------------------
# Orchestrator 1: BootstrapOrchestrator (bootstrap + discover modes)
# ---------------------------------------------------------------------------

import logging

logger = logging.getLogger(__name__)

_DEFAULT_SKILLS_DIR = Path(__file__).parent / "skills"


class BootstrapOrchestrator(Agent, llm=FakeLLMClient()):
    """Orchestrates the bootstrap/discover pipeline:
    Phase 1 Discovery → Phase 2 Solve → Phase 3 Verify/Reflect → Phase 4 Extract → save skills

    bootstrap mode: has ground truth, does reflection retries
    discover mode:  no ground truth, solve once then extract
    """

    DiscoveryAgent = DiscoveryAgent
    SolverAgent = SolverAgent
    SolutionVerifier = SolutionVerifier
    SkillExtractor = SkillExtractor

    MAX_RETRIES: int = 3

    def __init__(
        self,
        llm: "UnifiedLLM | None" = None,
        data: DataContext | None = None,
        skills_dir: Path | str | None = None,
        mode: Literal["bootstrap", "discover"] = "bootstrap",
        discovery_terms: int = 3,
        max_retries: int = 3,
        **kwargs,
    ):
        super().__init__(llm=llm, **kwargs)
        self.data = data or DataContext()
        self.skills_dir = Path(skills_dir) if skills_dir else _DEFAULT_SKILLS_DIR
        self.mode = mode
        self.discovery_terms = discovery_terms
        self.max_retries = max_retries

    def _format_output(self, result: "SolverOutput") -> str:
        """Format as FINAL ANSWER + JSON blob (same interface as original solver)."""
        payload = {
            "answer": result.answer,
            "correct": result.correct,
            "new_functions": result.new_functions,
            "new_example": result.new_example,
            "reflections": result.reflections,
            "solution_script": result.solution_script,
            "attempts": result.attempts,
        }
        return f"FINAL ANSWER: {result.answer}\n\n__STRUCTURED_OUTPUT__\n{json.dumps(payload, indent=2)}"

    async def run(self, inp: "SolverInput") -> str:
        """Run the full bootstrap/discover pipeline for one task.

        Returns formatted output string.
        """
        library = SkillLibrary(self.skills_dir)

        # Phase 1: Discovery
        discovery_agent = self.DiscoveryAgent(
            llm=self._llm,
            data=self.data,
            deep=True,
            discovery_terms=self.discovery_terms,
        )
        logger.info(f"[{self.mode}] Running discovery...")
        discovery_context = await discovery_agent.run(inp.question)
        logger.info(f"Discovery context: {len(discovery_context)} chars")

        # Phase 2+3: Solve (with optional reflection retries for bootstrap)
        verifier = self.SolutionVerifier(
            llm=self._llm,
            data=self.data,
            gt_answer=inp.gt_answer,
        )

        hint = ""
        reflections: list[str] = []
        result: ComputedAnswer | None = None
        is_correct = False
        max_attempts = (1 + self.max_retries) if self.mode == "bootstrap" and inp.gt_answer else 1

        for attempt in range(max_attempts):
            logger.info(f"[{self.mode}] Solve attempt {attempt + 1}/{max_attempts}")

            # Build solver with current context
            solver = self.SolverAgent(
                llm=self._llm,
                data=self.data,
                discovery_context=discovery_context,
                worked_examples=inp.existing_examples,
                helper_functions=inp.existing_functions,
            )
            solver.context.set("discovery_context", "self.discovery_context")
            solver.context.set("worked_examples", "self.worked_examples")

            result = await solver.compute_answer(inp.question, inp.guidelines, hint)

            # Phase 3: Verify (bootstrap only)
            if self.mode == "bootstrap" and inp.gt_answer:
                verification = await verifier.verify(
                    inp.question, inp.guidelines, result.answer, result.explanation
                )
                is_correct = verification.accepted
                logger.info(f"Attempt {attempt + 1}: correct={is_correct}")
                if is_correct:
                    break
                hint = verification.reflection
                reflections.append(
                    f"Attempt {attempt + 1}: Got '{result.answer}', "
                    f"expected '{inp.gt_answer}'. {hint[:100]}"
                )

        answer_str = str(result.answer) if result and result.answer is not None else ""

        # Phase 4: Extract skills
        work_summary = result.explanation if result else ""
        extractor = self.SkillExtractor(
            llm=self._llm,
            data=self.data,
            extract_solution_script=(self.mode == "bootstrap"),
        )
        extraction = await extractor.extract(
            question=inp.question,
            guidelines=inp.guidelines,
            gt_answer=inp.gt_answer if self.mode == "bootstrap" else "",
            work_summary=work_summary,
            reflections=reflections,
        )

        # Save to skill library
        library.save(extraction)
        logger.info(f"Skills saved to {self.skills_dir}")

        output = SolverOutput(
            answer=answer_str,
            correct=is_correct if self.mode == "bootstrap" else None,
            new_functions=extraction.helper_functions,
            new_example=(
                f"EXAMPLE - {extraction.example_name}:\n"
                f"Q: {extraction.example_question_pattern}\n"
                f"Rule: {extraction.example_rule}\n"
                f"A:\n{extraction.example_code.strip()}"
                if extraction.example_code.strip() else ""
            ),
            solution_script=str(extraction.solution_script) if extraction.solution_script else "",
            reflections=reflections,
            attempts=len(reflections) + 1,
        )
        return self._format_output(output)

    @no_trace
    async def _run_evaluation(self, task_input: dict) -> dict:
        """Entry point for evaluation harness compatibility."""
        try:
            inp = SolverInput(
                question=task_input.get("question", ""),
                gt_answer=str(task_input.get("answer", "")),
                guidelines=task_input.get("guidelines", ""),
                existing_functions=task_input.get("existing_functions", ""),
                existing_examples=task_input.get("existing_examples", ""),
            )
            if not inp.question:
                return {"response": "", "success": False, "error": "No question"}
            # Load data from data_dir if provided
            data_dir = task_input.get("data_dir", "")
            if data_dir:
                self.data = DataContext.from_dir(data_dir)
            output = await self.run(inp)
            answer = output.split("FINAL ANSWER:", 1)[-1].split("\n")[0].strip()
            return {"response": answer, "success": True, "answer": answer}
        except Exception as e:
            logger.error(f"BootstrapOrchestrator error: {e}", exc_info=True)
            return {"response": "", "success": False, "error": str(e)}

# ---------------------------------------------------------------------------
# Orchestrator 2: BenchmarkSolver (benchmark mode)
# ---------------------------------------------------------------------------


class BenchmarkSolver(Agent, llm=FakeLLMClient()):
    """Runs benchmark mode: load skills → discover → solve → return FINAL ANSWER.

    No skill extraction. No ground truth verification. Just solve as accurately as possible
    using the accumulated skill library.
    """

    DiscoveryAgent = DiscoveryAgent
    SolverAgent = SolverAgent

    def __init__(
        self,
        llm: "UnifiedLLM | None" = None,
        data: DataContext | None = None,
        skills_dir: Path | str | None = None,
        discovery_terms: int = 3,
        **kwargs,
    ):
        super().__init__(llm=llm, **kwargs)
        self.data = data or DataContext()
        self.skills_dir = Path(skills_dir) if skills_dir else _DEFAULT_SKILLS_DIR
        self.discovery_terms = discovery_terms

    def _parse_input(self, input_str: str) -> SolverInput:
        """Parse plain string or JSON dict into SolverInput."""
        try:
            data = json.loads(input_str.strip())
            return SolverInput(
                question=data.get("question", input_str),
                guidelines=data.get("guidelines", ""),
                existing_functions=data.get("existing_functions", ""),
                existing_examples=data.get("existing_examples", ""),
            )
        except (json.JSONDecodeError, TypeError):
            return SolverInput(question=str(input_str).strip())

    async def run(self, input_str: str) -> str:
        """Solve one benchmark task. Returns 'FINAL ANSWER: {answer}'."""
        inp = self._parse_input(input_str)

        # Load skill library
        library = SkillLibrary(self.skills_dir)
        helper_functions, worked_examples = library.load()

        # If caller didn't supply, use library defaults
        if not inp.existing_functions:
            inp = inp.model_copy(update={"existing_functions": helper_functions})
        if not inp.existing_examples:
            inp = inp.model_copy(update={"existing_examples": worked_examples})

        # Phase 1: Discovery (light — no deep schema exploration)
        discovery_agent = self.DiscoveryAgent(
            llm=self._llm,
            data=self.data,
            deep=False,
            discovery_terms=self.discovery_terms,
        )
        logger.info("[benchmark] Running discovery...")
        discovery_context = await discovery_agent.run(inp.question)

        # Phase 2: Solve
        solver = self.SolverAgent(
            llm=self._llm,
            data=self.data,
            discovery_context=discovery_context,
            worked_examples=inp.existing_examples,
            helper_functions=inp.existing_functions,
        )
        solver.context.set("discovery_context", "self.discovery_context")
        solver.context.set("worked_examples", "self.worked_examples")

        result = await solver.compute_answer(inp.question, inp.guidelines)
        answer_str = str(result.answer) if result.answer is not None else ""
        return f"FINAL ANSWER: {answer_str}"

    @no_trace
    async def _run_evaluation(self, task_input: dict) -> dict:
        """Entry point for evaluation harness compatibility."""
        try:
            input_str = (
                task_input
                if isinstance(task_input, str)
                else json.dumps({
                    "question": task_input.get("question", ""),
                    "guidelines": task_input.get("guidelines", ""),
                })
            )
            # Load data from data_dir if provided
            data_dir = task_input.get("data_dir", "") if isinstance(task_input, dict) else ""
            if data_dir:
                self.data = DataContext.from_dir(data_dir)
            output = await self.run(input_str)
            answer = output.replace("FINAL ANSWER:", "").strip()
            return {"response": answer, "success": True, "answer": answer}
        except Exception as e:
            logger.error(f"BenchmarkSolver error: {e}", exc_info=True)
            return {"response": "", "success": False, "error": str(e)}
```

#!/usr/bin/env python3
"""
CLI entry point for DABstep Solver.

Usage:
  # Benchmark mode (uses accumulated skill library):
  python run_solver.py --mode benchmark --input '{"question": "What is the total fee?"}' \\
      --data-dir /path/to/dabstep/data

  # Bootstrap mode (labeled task, accumulates skills):
  python run_solver.py --mode bootstrap \\
      --input '{"question": "...", "answer": "42.5", "guidelines": "..."}' \\
      --data-dir /path/to/dabstep/data

  # Discover mode (unlabeled task, accumulates skills):
  python run_solver.py --mode discover \\
      --input '{"question": "...", "guidelines": "..."}' \\
      --data-dir /path/to/dabstep/data
"""

import argparse
import asyncio
import json
import logging
import sys
from pathlib import Path

from dotenv import load_dotenv

# Ensure project root is on path
_here = Path(__file__).resolve().parent
_repo_root = _here.parent.parent
sys.path.insert(0, str(_repo_root))
load_dotenv(_repo_root / ".env")

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(name)s %(levelname)s %(message)s")

from dabstep_solver import BenchmarkSolver, BootstrapOrchestrator, DataContext  # noqa: E402


async def main() -> None:
    parser = argparse.ArgumentParser(description="DABstep Solver Agent")
    parser.add_argument(
        "--mode",
        choices=["bootstrap", "discover", "benchmark"],
        default="benchmark",
        help="Operation mode",
    )
    parser.add_argument("--input", required=True, help="JSON string or plain question text")
    parser.add_argument("--data-dir", default="", help="Path to DABstep data directory")
    parser.add_argument("--model", default="", help="LLM model name (e.g. aws/anthropic/bedrock-claude-sonnet-4-5-v1)")
    parser.add_argument("--provider", default="nvidia_internal", help="LLM provider")
    parser.add_argument("--skills-dir", default=str(_here / "skills"), help="Path to skills directory")
    parser.add_argument("--max-retries", type=int, default=3)
    parser.add_argument("--discovery-terms", type=int, default=3)
    args = parser.parse_args()

    # Load data
    data = DataContext.from_dir(args.data_dir) if args.data_dir else DataContext()

    # Build LLM client
    from unifiedllm import CompletionClient
    llm = CompletionClient(model=args.model, provider=args.provider) if args.model else None

    skills_dir = Path(args.skills_dir)

    if args.mode == "benchmark":
        agent = BenchmarkSolver(
            llm=llm,
            data=data,
            skills_dir=skills_dir,
            discovery_terms=args.discovery_terms,
        )
        output = await agent.run(args.input)
    else:
        try:
            inp_data = json.loads(args.input)
        except json.JSONDecodeError:
            inp_data = {"question": args.input}

        agent = BootstrapOrchestrator(
            llm=llm,
            data=data,
            skills_dir=skills_dir,
            mode=args.mode,
            max_retries=args.max_retries,
            discovery_terms=args.discovery_terms,
        )
        from dabstep_solver import SolverInput
        inp = SolverInput(
            question=inp_data.get("question", args.input),
            gt_answer=str(inp_data.get("answer", "")),
            guidelines=inp_data.get("guidelines", ""),
        )
        output = await agent.run(inp)

    print(output)


if __name__ == "__main__":
    asyncio.run(main())
```
